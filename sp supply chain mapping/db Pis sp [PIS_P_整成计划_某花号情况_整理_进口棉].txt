			成件时间=null,
			抽检时间=null,
			入库数量=total,
			单位,
			最早入库时间=null,
			最后入库时间=null,
			米制数量=cast(case when 单位='YD' then total*0.9144 else total end as numeric(18,2))
		from #印染成品 a
	end
	else
	begin
		select a.花号,
			开始检验时间=min(a.开始检验时间),
			结束检验时间=max(a.结束检验时间),
			成件时间=max(a.成件时间),
			抽检时间=max(b.抽检时间),
			入库数量=sum(a.入库数量),
			a.单位,
			最早入库时间=min(a.最早入库时间),
			最后入库时间=max(a.最后入库时间),
			米制数量=sum(case when a.单位='YD' then cast(a.入库数量*0.9144 as numeric(9,2)) else a.入库数量 end)
		from #仓库基础数据B a
		left join #成品抽检数据 b
		on a.客单号=b.客单号 and a.客单行号=b.客单行号 and a.分批号=b.分批号
		group by a.花号,a.单位
		union
		select 花号=a.花号+a.成件编号,
			开始检验时间=null,
			结束检验时间=null,
			成件时间=null,
			抽检时间=null,
			入库数量=sum(a.total),
			a.单位,
			最早入库时间=null,
			最后入库时间=null,
			米制数量=sum(cast(case when 单位='YD' then total*0.9144 else total end as numeric(18,2)))
		from #印染成品 a
		group by a.花号+a.成件编号,a.单位
	end
