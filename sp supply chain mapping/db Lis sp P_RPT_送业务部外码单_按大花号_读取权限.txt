create   PROC [dbo].[P_RPT_送业务部外码单_按大花号_读取权限]
    (
      @客单号 CHAR(6) ,
      @type CHAR(1) = NULL ,      --是否显示等级品
      @开始件号 INT = NULL ,
      @结束件号 INT = NULL
    )
AS





declare @品名 varchar(50),@织厂类别 varchar(50)

select top 1 @织厂类别=织厂类别,@品名=品名 from mycrmlf.dbo.v_OrdOfAllSts
where 客单号=@客单号  

if @客单号 in ('515015','539922')
begin
set @织厂类别='联发'
end

	if @织厂类别='印染'  and @品名  like '%色织%印花%' 
	begin
		set @织厂类别='联发'
	end
	else
	begin	
		print '1'
	end

SELECT 织厂类别=@织厂类别
