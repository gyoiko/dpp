sp P_进口棉_原棉产地查询
SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO
CREATE proc [dbo].[P_进口棉_原棉产地查询] 
(@客单号 varchar(50))
as
begin
  declare @织厂类别 varchar(50)
  select @织厂类别=(select top 1 织厂类别 from [mycrmlf].[dbo].v_OrdOfAllSts with(nolock) where 客单号=@客单号)
  if(@织厂类别='印染')
  begin
    select 客单号,key1,key2,key3,批号 
    into #操作明细
    from lfpd.[dbo].[T_仓库_操作明细表]   
    where 物料类型=1 and 客单号=@客单号
  
    select a.客单号,a.key1,a.key2,a.key3,原棉产地=b.原产地,纱线产地,a.批号 
    from #操作明细 a
    inner join lfpd.[dbo].[T_仓库_操作明细表] b on a.key1=b.key1 and a.key2=b.key2 and a.key3=b.key3 and a.批号=b.批号  and b.转仓事务代码='51'
    inner join [tesdatt5].dbo.T_仓库_采购单表 c on b.采购单号=c.采购单号 and b.行号=c.行号 
  end
  else begin
    select distinct a.客单号,b.key1,b.key2,b.key3,原棉产地=b.原产地,b.纱线产地,b.批号,纱厂名称=b.备注
    from [tesdatt5].[dbo].[T_仓库_操作明细表] a
    left join [tesdatt5].[dbo].[T_仓库_物料库存表] b on a.key1=b.key1 and a.key2=b.key2 and a.key3=b.key3 and a.key4=b.key4 and a.批号=b.批号
    where a.客单号=@客单号
  end
end

@类别 VARCHAR(20),
@国别 VARCHAR(50),
@客单号 VARCHAR(20),
@客单行 INT,
@lang int = 1 
)
AS
BEGIN
	SET NOCOUNT ON;
	/*
	SELECT TOP 1000 * FROM mycrmlf.DBO.V_OrderKeDanJia
	SELECT * FROM T_进口棉_配棉主表
	SELECT * FROM T_进口棉_配棉登记表
	
	DECLARE @类别 VARCHAR(20),@国别 VARCHAR(50),@客单号 VARCHAR(20),@客单行 INT
	SET @类别=''
	SET @国别=''
	SET @客单号='479983'
	SET @客单行='3'	
    --*/

	 IF OBJECT_ID('tempdb..#contracts_stage') IS NOT NULL
        DROP TABLE #contracts_stage;

	 CREATE TABLE #contracts_stage (
        客单号 VARCHAR(20),
        客单行 INT,
        订单数量 DECIMAL(18,2),
        订单单位 NVARCHAR(50),
        类别 NVARCHAR(200),     
        国别 NVARCHAR(200),    
        合同号 NVARCHAR(100),
        批号 NVARCHAR(100),
        文件描述 NVARCHAR(500),
        文件明细 NVARCHAR(MAX),
        纱支 NVARCHAR(100),
        使用数量 DECIMAL(18,2),
        使用单位 NVARCHAR(50),
        品名 NVARCHAR(500),     
        经纱 NVARCHAR(200),
        纬纱 NVARCHAR(200)
    );
	INSERT INTO #contracts_stage
    SELECT B.客单号,B.客单行,订单数量=C.数量,订单单位=C.单位		
		,A.类别,国别,合同号
		--,合同数量
		,批号
		--,可用数量
		,文件描述,文件明细
		--,入库时间
		--,发票数量		
		,B.纱支,使用数量=B.数量,使用单位=B.单位		
		,C.品名,C.经纱,C.纬纱
    FROM T_进口棉_配棉主表 A WITH(NOLOCK)
    INNER JOIN T_进口棉_配棉登记表 B WITH(NOLOCK)
    ON A.ID=B.MainID AND B.删除标志=0
    INNER JOIN mycrmlf.DBO.V_OrderKeDanJia C
    ON B.客单号=C.客单号 AND B.客单行=C.客单行
    WHERE A.删除标志=0
		AND A.类别=CASE WHEN @类别='' THEN A.类别 ELSE @类别 END
		AND A.国别=CASE WHEN @国别='' THEN A.国别 ELSE @国别 END
		AND ISNULL(B.客单号,'')=CASE WHEN @客单号='' THEN ISNULL(B.客单号,'') ELSE @客单号 END
		AND ISNULL(B.客单行,'')=CASE WHEN @客单号='' OR @客单行='' THEN ISNULL(B.客单行,'') ELSE @客单行 END
    
	
	IF @lang <>0  
    BEGIN
        UPDATE #contracts_stage
        SET 
            类别 = ISNULL(pdpis.dbo.fn_GetTranslation(类别, @lang), 类别),
            国别 = ISNULL(pdpis.dbo.fn_GetTranslation(国别, @lang), 国别),
			品名 = ISNULL(pdpis.dbo.fn_GetTranslation(品名, @lang), 品名),

			文件明细 = ISNULL(pdpis.dbo.fn_GetTranslation(文件明细, @lang), 文件明细)

		 END

    SELECT * FROM #contracts_stage;


    IF OBJECT_ID('tempdb..#contracts_stage') IS NOT NULL
        DROP TABLE #contracts_stage;
END

	select ID,文件名称,附件名称,文件打开类型,文件类别,提交人,提交时间

GO
SET QUOTED_IDENTIFIER ON
GO
-- =============================================
-- Author:		<Author,,Name>
-- Create date: <Create Date,,>
-- Description:	<Description,,>
-- Exec [P_进口棉_配棉核销明细查询] 114
-- =============================================
CREATE PROCEDURE [dbo].[P_进口棉_配棉核销明细查询](
@ID INT,
@lang int = 0
)
AS
BEGIN
	SET NOCOUNT ON;

	--Translation
	 IF OBJECT_ID('tempdb..#results1') IS NOT NULL DROP TABLE #results1
	 SELECT 
        ID,
        MainID,
        客单号,
        客单行,
        类别,      
        纱支,
        数量,
        单位,
        新增人,
        新增时间,
        品名,      
        经纱,
        纬纱
    INTO #results1
    FROM (
	--SELECT TOP 1000 * FROM mycrmlf.DBO.v_OrdOfAllSts WITH(NOLOCK)
    
    SELECT ID,MainID,A.客单号,A.客单行,A.类别,A.纱支,A.数量,A.单位,A.新增人,A.新增时间,B.品名,B.经纱,B.纬纱
    FROM T_进口棉_配棉登记表 A WITH(NOLOCK)
    LEFT JOIN mycrmlf.DBO.v_OrdOfAllSts B WITH(NOLOCK)
    ON A.客单号=B.客单号 AND A.客单行=B.客单行
    WHERE A.删除标志=0 AND MainID=@ID
    ) AS SourceData

	UPDATE #results1
    SET 
        类别 = pdpis.dbo.fn_GetTranslation(类别, @lang),    
        品名 = pdpis.dbo.fn_GetTranslation(品名, @lang)

	SELECT 
        ID,
        MainID,
        客单号,
        客单行,
        类别,     
        纱支,
        数量,
        单位,
        新增人,
        新增时间,
        品名,      
        经纱,
        纬纱
    FROM #results1
    
    -- Cleanup
    IF OBJECT_ID('tempdb..#results1') IS NOT NULL DROP TABLE #results1
END
